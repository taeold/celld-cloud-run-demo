import { WorkflowEntrypoint } from "cloudflare:workers";

export class DemoWorkflow extends WorkflowEntrypoint {
async run(event, step) {
const init = await step.do("fetch-json", () =>
fetch("https://httpbin.org/json"));
const [delayed, bytes, post] = await Promise.all([
step.do("fetch-delay-1s", () =>
fetch("https://httpbin.org/delay/1")),
step.do("fetch-bytes", () =>
fetch("https://httpbin.org/bytes/16384")),
step.do("post-json", () =>
fetch("https://httpbin.org/post", { method: "POST" })),
]);
await step.sleep("sleep-3s", "3 seconds");
const uuid = await step.do("fetch-uuid", () =>
fetch("https://httpbin.org/uuid"));
return await step.do("finish", () => ({
status: "ok", uuid: uuid.uuid
}));
}
}
